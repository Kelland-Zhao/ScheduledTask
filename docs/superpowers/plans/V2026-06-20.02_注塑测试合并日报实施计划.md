> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

function _itr_normalizeDate(value) {
  if (!value) return "";
  if (value instanceof Date && !isNaN(value.getTime())) {
    return Utilities.formatDate(value, ITR_CONFIG.TIME_ZONE, "yyyy-MM-dd");
  }
  var text = String(value).trim();
  var match = text.match(/^(\d{4})[\/-](\d{1,2})[\/-](\d{1,2})/);
  if (match) {
    return match[1] + "-" + ("0" + match[2]).slice(-2) + "-" + ("0" + match[3]).slice(-2);
  }
  var date = new Date(text);
  return isNaN(date.getTime())
    ? ""
    : Utilities.formatDate(date, ITR_CONFIG.TIME_ZONE, "yyyy-MM-dd");
}

function _itr_uniqueEmails(values) {
  var seen = {};
  return (values || []).map(function(value) {
    return String(value || "").trim().toLowerCase();
  }).filter(function(email) {
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) || seen[email]) return false;
    seen[email] = true;
    return true;
  });
}

function _itr_escapeHtml(value) {
  return String(value == null ? "" : value)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}
```

function _itr_getRecordsByDate(sheet, targetDateKey) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  var range = sheet.getRange(2, 1, lastRow - 1, 21);
  var values = range.getValues();
  var richText = range.getRichTextValues();
  var result = [];
  for (var i = 0; i < values.length; i++) {
    if (_itr_normalizeDate(values[i][1]) !== targetDateKey) continue;
    result.push({
      rowNumber: i + 2,
      values: values[i],
      links: {
        sample: _itr_getRichTextLink(richText[i][16]),
        record: _itr_getRichTextLink(richText[i][17])
      }
    });
  }
  return result;
}

function _itr_getRichTextLink(richTextValue) {
  if (!richTextValue) return "";
  var direct = richTextValue.getLinkUrl();
  if (direct) return direct;
  var runs = richTextValue.getRuns();
  for (var i = 0; i < runs.length; i++) {
    var link = runs[i].getLinkUrl();
    if (link) return link;
  }
  return "";
}

function _itr_fillSmartChipLinks(records, sheetName) {
  var targets = [];
  records.forEach(function(record) {
    if (record.values[16] && !record.links.sample) {
      targets.push({ record: record, key: "sample", a1: "Q" + record.rowNumber });
    }
    if (record.values[17] && !record.links.record) {
      targets.push({ record: record, key: "record", a1: "R" + record.rowNumber });
    }
  });
  targets.forEach(function(target) {
    try {
      var response = Sheets.Spreadsheets.get(ITR_CONFIG.SPREADSHEET_ID, {
        ranges: ["'" + sheetName.replace(/'/g, "''") + "'!" + target.a1],
        fields: "sheets(data(rowData(values(chipRuns))))"
      });
      var value = response.sheets &&
        response.sheets[0].data &&
        response.sheets[0].data[0].rowData &&
        response.sheets[0].data[0].rowData[0].values &&
        response.sheets[0].data[0].rowData[0].values[0];
      var runs = value && value.chipRuns ? value.chipRuns : [];
      for (var i = 0; i < runs.length; i++) {
        var uri = runs[i].chip &&
          runs[i].chip.richLinkProperties &&
          runs[i].chip.richLinkProperties.uri;
        if (uri) {
          target.record.links[target.key] = uri;
          break;
        }
      }
    } catch (error) {
      console.warn("Smart Chip解析失败 " + target.a1 + ": " + error.message);
    }
  });
}

function _itr_getRecipients(spreadsheet, records) {
  var emails = [];
  var notificationSheet = spreadsheet.getSheetByName(ITR_CONFIG.NOTIFICATION_SHEET);
  if (!notificationSheet) throw new Error("未找到工作表: " + ITR_CONFIG.NOTIFICATION_SHEET);

  if (notificationLastRow >= 2) {
    notificationSheet.getRange(2, 1, notificationLastRow - 1, 2).getDisplayValues()
      .forEach(function(row) {
        if (row[0] && (row[1].indexOf("测试后日报") !== -1 || row[1].indexOf("测试前日报") !== -1)) {
          emails.push(row[0]);
        }
      });
  }

  var managerNames = {};
  records.forEach(function(record) {
    var name = String(record.values[12] || "").trim();
    if (name) managerNames[name] = true;
  });

  var nameSheet = spreadsheet.getSheetByName(ITR_CONFIG.NAME_SHEET);

function _itr_run(options) {
  var now = options.now || new Date();
  var spreadsheet = SpreadsheetApp.openById(ITR_CONFIG.SPREADSHEET_ID);
  var sheet = spreadsheet.getSheetByName(ITR_CONFIG.TEST_SHEET);
  if (!sheet) throw new Error("未找到工作表: " + ITR_CONFIG.TEST_SHEET);

  var yesterdayKey = _itr_dateKey(_itr_addDays(now, -1));
  var tomorrowKey = _itr_dateKey(_itr_addDays(now, 1));
  var afterRecords = _itr_getRecordsByDate(sheet, yesterdayKey);
  var beforeRecords = _itr_getRecordsByDate(sheet, tomorrowKey);

  if (afterRecords.length === 0 && beforeRecords.length === 0) {
    return { skipped: true, yesterday: 0, tomorrow: 0, problems: 0, recipients: 0 };
  }

  afterRecords.forEach(function(record) {
    record.problems = _itr_checkAfterProblems(record);
  });
  beforeRecords.forEach(function(record) {
    record.problems = _itr_checkBeforeProblems(record);
  });
  _itr_fillSmartChipLinks(afterRecords.concat(beforeRecords), ITR_CONFIG.TEST_SHEET);

  var recipients = options.testMode
    ? [ITR_CONFIG.TEST_EMAIL]
    : _itr_getRecipients(spreadsheet, afterRecords.concat(beforeRecords));
  if (recipients.length === 0) throw new Error("未找到有效收件人");

  var reportDate = _itr_dateKey(now);
  var html = _itr_buildEmailHtml({
    reportDate: reportDate,
    yesterdayKey: yesterdayKey,
    tomorrowKey: tomorrowKey,
    afterRecords: afterRecords,
    beforeRecords: beforeRecords
  });
  var subject = (options.testMode ? "[测试]" : "") +
    "【注塑测试日报】" + reportDate + " 昨日复盘 & 明日提醒";
  _itr_sendEmail(recipients, subject, html);

  return {
    skipped: false,
    yesterday: afterRecords.length,
    tomorrow: beforeRecords.length,
    problems: afterRecords.concat(beforeRecords).filter(function(record) {
      return record.problems.length > 0;
    }).length,
    recipients: recipients.length
  };
}
```
